# The window, the font and the input handling - the only place pygame is
# allowed to appear is render/screen.py and main.py.
#
# pygame-ce, NOT pygame. They are rival forks that both install a module
# called `pygame`, so having both in one environment gives you whichever
# happens to win. This is the fork that is maintained, and it is what the
# game is built and tested against; `pip install pygame` alongside it is the
# one mistake this file exists to prevent.
pygame-ce>=2.5,<3

# Tile arrays. Chunks are (64, 64) int8 grids and the generators do their
# flood fills and erosion passes on them wholesale.
numpy>=1.24,<3

pytest>=8,<10
